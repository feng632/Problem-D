\makeatletter
\newenvironment{bibliographywithouttitle}[1]
  {\list{\@biblabel{\@arabic\c@enumiv}}%
       {\settowidth\labelwidth{\@biblabel{#1}}%
        \leftmargin\labelwidth
        \advance\leftmargin\labelsep
        \@openbib@code
        \usecounter{enumiv}%
        \let\p@enumiv\@empty
        \renewcommand\theenumiv{\@arabic\c@enumiv}}%
   \sloppy
   \clubpenalty4000
   \@clubpenalty \clubpenalty
   \widowpenalty4000%
   \sfcode`\.=\@m}
  {\def\@noitemerr
     {\@latex@warning{Empty `bibliographywithouttitle' environment}}%
   \endlist}
\makeatother
